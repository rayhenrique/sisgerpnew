-- Habilita UUIDs (opcional, mas recomendado no Supabase)
create extension if not exists "uuid-ossp";

-- Tabela de Categorias (Recriada para Postgres)
CREATE TABLE public.categories (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name text NOT NULL,
    code text,
    type text NOT NULL, -- fonte, bloco, grupo, acao
    parent_id bigint REFERENCES public.categories(id),
    active boolean DEFAULT true,
    description text,
    created_at timestamp with time zone DEFAULT now(),
    updated_at timestamp with time zone DEFAULT now()
);

-- Tabela de Classificação de Despesas
CREATE TABLE public.expense_classifications (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name text NOT NULL,
    code text UNIQUE,
    description text,
    active boolean DEFAULT true,
    created_at timestamp with time zone DEFAULT now(),
    updated_at timestamp with time zone DEFAULT now()
);

-- Tabela de Receitas (Revenues)
CREATE TABLE public.revenues (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    description text NOT NULL,
    amount numeric(15,2) NOT NULL,
    date date NOT NULL,
    fonte_id bigint REFERENCES public.categories(id),
    bloco_id bigint REFERENCES public.categories(id),
    grupo_id bigint REFERENCES public.categories(id),
    acao_id bigint REFERENCES public.categories(id),
    observation text,
    created_at timestamp with time zone DEFAULT now(),
    updated_at timestamp with time zone DEFAULT now()
);

-- Tabela de Despesas (Expenses)
CREATE TABLE public.expenses (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    description text NOT NULL,
    amount numeric(15,2) NOT NULL,
    date date NOT NULL,
    fonte_id bigint REFERENCES public.categories(id),
    bloco_id bigint REFERENCES public.categories(id),
    grupo_id bigint REFERENCES public.categories(id),
    acao_id bigint REFERENCES public.categories(id),
    expense_classification_id bigint REFERENCES public.expense_classifications(id),
    observation text,
    created_at timestamp with time zone DEFAULT now(),
    updated_at timestamp with time zone DEFAULT now()
);

-- Configurações da Cidade
CREATE TABLE public.city_settings (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    city_name text NOT NULL,
    city_hall_name text NOT NULL,
    address text NOT NULL,
    ibge_code text NOT NULL,
    state text NOT NULL,
    zip_code text,
    phone text,
    email text,
    mayor_name text,
    created_at timestamp with time zone DEFAULT now(),
    updated_at timestamp with time zone DEFAULT now()
);
